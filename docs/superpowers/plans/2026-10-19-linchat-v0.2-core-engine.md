# LinChat v0.2 Core Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the headless core of linchat v0.2: multi-provider LLM layer (thin `requests` clients with streaming + tool calls), config v2 with env discovery and migration, CLI-agent bridges, provider registry, and slash-command parser.

**Architecture:** The single-file `linchat.py` becomes a package. Providers implement one protocol — `stream(messages, tools) -> Iterator[Event]` — emitting `TextDelta`/`ToolCall`/`Usage`/`Done`/`ErrorEvent`. Agent CLIs (opencode, claude, codex, gemini, pi, crush) are providers backed by subprocess. No GTK code in this plan; the existing `linchat.py` UI keeps working unchanged until Plan 2 wires the new engine in.

**Tech Stack:** Python 3.10+, `requests` (HTTP + SSE), stdlib only otherwise (`configparser`, `subprocess`, `shlex`, `json`, `dataclasses`). pytest for tests.

**Spec:** `docs/superpowers/specs/2026-10-19-linchat-v0.2-design.md` — sections "Architecture", "Providers", "Config". This plan implements the headless core; tools/UI/packaging follow in Plans 2 and 3.

## Global Constraints

- Python >= 3.10 (uses `X | Y` unions, dataclasses, match-free code)
- Runtime deps unchanged: `requests`, `markdown`, `PyGObject`. **No new runtime pip deps.** pytest is dev-only.
- No vendor SDKs, no LiteLLM — all HTTP via `requests`, SSE parsed by hand.
- GTK main thread never blocks; providers are called from worker threads (irrelevant to this plan's code, but all provider APIs must be thread-safe to *call*).
- Existing `linchat.py` must keep running at every commit (it stays untouched until Plan 2).
- API keys never appear in error messages or logs.
- Version constant: `0.2.0` in `linchat/__init__.py`.
- Commit style: repo uses `feat:`, `fix:`, `ci:`, `pkg:`, `docs:` prefixes.

```python
"""Provider protocol, event types, and SSE parsing helpers."""
from __future__ import annotations

import dataclasses
from typing import Any, Iterable, Iterator, Union


@dataclasses.dataclass
class TextDelta:
    text: str


@dataclasses.dataclass
class ToolCall:
    id: str
    name: str
    args: dict


@dataclasses.dataclass
class Usage:
    input_tokens: int = 0
    output_tokens: int = 0


@dataclasses.dataclass
class Done:
    reason: str = "stop"  # stop | tool_calls | max_hops | denied | error


@dataclasses.dataclass
class ErrorEvent:
    message: str


Event = Union[TextDelta, ToolCall, Usage, Done, ErrorEvent]


@dataclasses.dataclass
class Message:
    role: str  # user | assistant | tool
    content: Any = ""  # str, or provider-native list of parts
    tool_calls: list | None = None  # assistant turn: [{id, name, args}]
    tool_call_id: str | None = None  # tool turn: which call this answers
    name: str | None = None
    images: list[bytes] | None = None  # PNG bytes for vision providers


def parse_sse_lines(lines: Iterable[str]) -> Iterator[str]:
    """Yield one joined data payload per SSE event. Keeps '[DONE]' as-is."""
    data_lines: list[str] = []
    for raw in lines:
        line = raw.rstrip("\r\n")
        if line == "":
            if data_lines:
                yield "\n".join(data_lines)
                data_lines = []
            continue
        if line.startswith(":"):
            continue
        if line.startswith("data:"):
            data_lines.append(line[5:].lstrip(" "))
        # other fields (event:, id:, retry:) ignored
    if data_lines:
        yield "\n".join(data_lines)


def iter_sse(resp) -> Iterator[str]:
    """Parse SSE from a streamed requests.Response."""
    return parse_sse_lines(resp.iter_lines(decode_unicode=True))
```

"""OpenAI-compatible chat-completions client (OpenAI, Groq, OpenRouter,
DeepSeek, Mistral, xAI, LM Studio, llama.cpp, vLLM, Ollama-compat, ...)."""
from __future__ import annotations

import json
from typing import Iterator

import requests

from .base import (
    Done, ErrorEvent, Event, Message, TextDelta, ToolCall, Usage, iter_sse,
)


def messages_to_openai(messages: list[Message]) -> list[dict]:
    out: list[dict] = []
    for m in messages:
        if m.role == "tool":
            out.append({"role": "tool", "tool_call_id": m.tool_call_id or "", "content": str(m.content)})
        elif m.role == "assistant" and m.tool_calls:
            out.append({
                "role": "assistant",
                "content": m.content or "",
                "tool_calls": [{
                    "id": tc["id"], "type": "function",
                    "function": {"name": tc["name"], "arguments": json.dumps(tc["args"])},
                } for tc in m.tool_calls],
            })
        else:
            msg: dict = {"role": m.role, "content": m.content}
            if m.images:
                parts = [{"type": "text", "text": str(m.content) or ""}]
                for png in m.images:
                    import base64
                    parts.append({"type": "image_url",
                                  "image_url": {"url": "data:image/png;base64," + base64.b64encode(png).decode()}})
                msg["content"] = parts
            out.append(msg)
    return out


class OpenAICompatProvider:
    def __init__(self, name: str, api_key: str = "",
                 base_url: str = "https://api.openai.com/v1",
                 model: str = "", timeout: int = 60):
        self.name = name
        self.api_key = api_key
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.timeout = timeout

    def _headers(self) -> dict:
        h = {"Content-Type": "application/json"}
        if self.api_key:
            h["Authorization"] = f"Bearer {self.api_key}"
        return h

    def list_models(self) -> list[str]:
        try:
            r = requests.get(f"{self.base_url}/models", headers=self._headers(), timeout=self.timeout)
            r.raise_for_status()
            return sorted(m["id"] for m in r.json().get("data", []))
        except Exception:
            return []

    def stream(self, messages: list[Message], tools: list[dict] | None = None) -> Iterator[Event]:
        payload: dict = {
            "model": self.model,
            "messages": messages_to_openai(messages),
            "stream": True,
            "stream_options": {"include_usage": True},
        }
        if tools:
            payload["tools"] = tools
        try:
            resp = requests.post(
                f"{self.base_url}/chat/completions", headers=self._headers(),
                json=payload, timeout=self.timeout, stream=True,
            )
        except requests.RequestException as e:
            yield ErrorEvent(f"Connection error: {e.__class__.__name__}")
            return
        if resp.status_code != 200:
            yield ErrorEvent(f"HTTP {resp.status_code}: {resp.text[:300]}")
            return

        tool_acc: dict[int, dict] = {}  # index -> {id, name, args_str}
        saw_tool_calls = False
        try:
            for data in iter_sse(resp):
                if data == "[DONE]":
                    break
                try:
                    chunk = json.loads(data)
                except json.JSONDecodeError:
                    continue
                if chunk.get("usage"):
                    u = chunk["usage"]
                    yield Usage(u.get("prompt_tokens", 0), u.get("completion_tokens", 0))
                choices = chunk.get("choices") or []
                if not choices:
                    continue
                ch = choices[0]
                delta = ch.get("delta") or {}
                if delta.get("content"):
                    yield TextDelta(delta["content"])
                for tc in delta.get("tool_calls") or []:
                    idx = tc.get("index", 0)
                    acc = tool_acc.setdefault(idx, {"id": "", "name": "", "args": ""})
                    fn = tc.get("function") or {}
                    if tc.get("id"):
                        acc["id"] = tc["id"]
                    if fn.get("name"):
                        acc["name"] = fn["name"]
                    if fn.get("arguments"):
                        acc["args"] += fn["arguments"]
                finish = ch.get("finish_reason")
                if finish == "tool_calls" and tool_acc:
                    for idx in sorted(tool_acc):
                        acc = tool_acc[idx]
                        try:
                            args = json.loads(acc["args"] or "{}")
                        except json.JSONDecodeError:
                            args = {"_raw": acc["args"]}
                        yield ToolCall(acc["id"], acc["name"], args)
                        saw_tool_calls = True
                    tool_acc = {}
            yield Done("tool_calls" if saw_tool_calls else "stop")
        except requests.RequestException as e:
            yield ErrorEvent(f"Stream error: {e.__class__.__name__}")
```

"""Anthropic Messages API client (thin, requests-only)."""
from __future__ import annotations

import base64
import json
from typing import Iterator

import requests

from .base import (
    Done, ErrorEvent, Event, Message, TextDelta, ToolCall, Usage, iter_sse,
)

API_URL = "https://api.anthropic.com/v1"
VERSION = "2023-06-01"


def messages_to_anthropic(messages: list[Message]) -> list[dict]:
    out: list[dict] = []
    for m in messages:
        if m.role == "tool":
            block = {"type": "tool_result", "tool_use_id": m.tool_call_id or "", "content": str(m.content)}
            if out and out[-1]["role"] == "user" and isinstance(out[-1]["content"], list) \
                    and out[-1]["content"] and out[-1]["content"][-1]["type"] == "tool_result":
                out[-1]["content"].append(block)
            else:
                out.append({"role": "user", "content": [block]})
        elif m.role == "assistant" and m.tool_calls:
            blocks = []
            if m.content:
                blocks.append({"type": "text", "text": str(m.content)})
            for tc in m.tool_calls:
                blocks.append({"type": "tool_use", "id": tc["id"], "name": tc["name"], "input": tc["args"]})
            out.append({"role": "assistant", "content": blocks})
        else:
            blocks = [{"type": "text", "text": str(m.content) or ""}]

                    "type": "base64", "media_type": "image/png",
                    "data": base64.b64encode(png).decode()}})

class AnthropicProvider:
    def __init__(self, name: str, api_key: str, model: str = "", timeout: int = 60):
        self.name = name
        self.api_key = api_key
        self.model = model or "claude-sonnet-4"
        self.timeout = timeout

    def _post(self, payload: dict) -> requests.Response:
        return requests.post(
            f"{API_URL}/messages",
            headers={"x-api-key": self.api_key, "anthropic-version": VERSION,
                     "Content-Type": "application/json"},
            json=payload, timeout=self.timeout, stream=True,
        )

    def list_models(self) -> list[str]:
        try:
            r = requests.get(f"{API_URL}/models",
                             headers={"x-api-key": self.api_key, "anthropic-version": VERSION},
                             timeout=self.timeout)
            r.raise_for_status()
            return sorted(m["id"] for m in r.json().get("data", []))
        except Exception:
            return []

    def stream(self, messages: list[Message], tools: list[dict] | None = None) -> Iterator[Event]:
        payload: dict = {"model": self.model, "max_tokens": 8192, "stream": True,
                         "messages": messages_to_anthropic(messages)}
        if tools:
            payload["tools"] = tools
        try:
            resp = self._post(payload)
        except requests.RequestException as e:
            yield ErrorEvent(f"Connection error: {e.__class__.__name__}")
            return
        if resp.status_code != 200:
            yield ErrorEvent(f"HTTP {resp.status_code}: {resp.text[:300]}")
            return

        # index -> {type, id, name, args_json}
        blocks: dict[int, dict] = {}
        usage = Usage()
        stop_reason = "stop"
        try:
            for data in iter_sse(resp):
                try:
                    ev = json.loads(data)
                except json.JSONDecodeError:
                    continue
                et = ev.get("type")
                if et == "message_start":
                    usage.input_tokens = ev["message"].get("usage", {}).get("input_tokens", 0)
                elif et == "content_block_start":
                    cb = ev["content_block"]
                    if cb["type"] == "tool_use":
                        blocks[ev["index"]] = {"type": "tool_use", "id": cb["id"], "name": cb["name"], "json": ""}
                elif et == "content_block_delta":
                    d = ev["delta"]
                    if d["type"] == "text_delta":
                        yield TextDelta(d["text"])
                    elif d["type"] == "input_json_delta" and ev["index"] in blocks:
                        blocks[ev["index"]]["json"] += d.get("partial_json", "")
                elif et == "message_delta":
                    stop_reason = ev.get("delta", {}).get("stop_reason") or stop_reason
                    usage.output_tokens = ev.get("usage", {}).get("output_tokens", usage.output_tokens)
                elif et == "message_stop":
                    break
            yield usage
            for b in blocks.values():
                try:
                    args = json.loads(b["json"] or "{}")
                except json.JSONDecodeError:
                    args = {"_raw": b["json"]}
                yield ToolCall(b["id"], b["name"], args)
            yield Done("tool_calls" if blocks else "stop")
        except requests.RequestException as e:
            yield ErrorEvent(f"Stream error: {e.__class__.__name__}")

"""Google Gemini generateContent client (thin, requests-only)."""
from __future__ import annotations

import base64
import json
from typing import Iterator

import requests

from .base import (
    Done, ErrorEvent, Event, Message, TextDelta, ToolCall, Usage, iter_sse,
)

BASE = "https://generativelanguage.googleapis.com/v1beta"


def messages_to_gemini(messages: list[Message]) -> list[dict]:
    contents: list[dict] = []
    for m in messages:
        if m.role == "tool":
            part = {"functionResponse": {"name": m.name or m.tool_call_id or "tool",
                                         "response": {"result": str(m.content)}}}
            if contents and contents[-1]["role"] == "user" and \
                    contents[-1]["parts"] and "functionResponse" in contents[-1]["parts"][-1]:
                contents[-1]["parts"].append(part)
            else:
                contents.append({"role": "user", "parts": [part]})
        elif m.role == "assistant" and m.tool_calls:
            parts = []
            if m.content:
                parts.append({"text": str(m.content)})
            for i, tc in enumerate(m.tool_calls):
                parts.append({"functionCall": {"name": tc["name"], "args": tc["args"]}})
            contents.append({"role": "model", "parts": parts})
        else:
            role = "model" if m.role == "assistant" else "user"
            parts = [{"text": str(m.content) or ""}]
            for png in m.images or []:
                parts.append({"inlineData": {"mimeType": "image/png",
                                             "data": base64.b64encode(png).decode()}})
            contents.append({"role": role, "parts": parts})
    return contents


class GeminiProvider:
    def __init__(self, name: str, api_key: str, model: str = "", timeout: int = 60):
        self.name = name
        self.api_key = api_key
        self.model = model or "gemini-2.5-flash"
        self.timeout = timeout

    def list_models(self) -> list[str]:
        try:
            r = requests.get(f"{BASE}/models", params={"key": self.api_key}, timeout=self.timeout)
            r.raise_for_status()
            names = [m["name"].split("/")[-1] for m in r.json().get("models", [])
                     if "generateContent" in (m.get("supportedGenerationMethods") or [])]
            return sorted(names)
        except Exception:
            return []

    def stream(self, messages: list[Message], tools: list[dict] | None = None) -> Iterator[Event]:
        url = f"{BASE}/models/{self.model}:streamGenerateContent?alt=sse&key={self.api_key}"
        payload: dict = {"contents": messages_to_gemini(messages)}
        if tools:
            # Convert OpenAI-style tool specs to Gemini functionDeclarations.
            decls = []
            for t in tools:
                fn = t.get("function", t)
                decls.append({"name": fn["name"], "description": fn.get("description", ""),
                              "parameters": fn.get("parameters", {"type": "object", "properties": {}})})
            payload["tools"] = [{"functionDeclarations": decls}]
        try:
            resp = requests.post(url, json=payload, timeout=self.timeout, stream=True)
        except requests.RequestException as e:
            yield ErrorEvent(f"Connection error: {e.__class__.__name__}")
            return
        if resp.status_code != 200:
            yield ErrorEvent(f"HTTP {resp.status_code}: {resp.text[:300]}")
            return

        usage = Usage()
        saw_calls = False
        call_idx = 0
        try:
            for data in iter_sse(resp):
                try:
                    chunk = json.loads(data)
                except json.JSONDecodeError:
                    continue
                meta = chunk.get("usageMetadata") or {}
                if meta:
                    usage = Usage(meta.get("promptTokenCount", 0), meta.get("candidatesTokenCount", 0))
                for cand in chunk.get("candidates", []):
                    for part in (cand.get("content") or {}).get("parts", []):
                        if "text" in part:
                            yield TextDelta(part["text"])
                        if "functionCall" in part:
                            fc = part["functionCall"]
                            saw_calls = True
                            yield ToolCall(f"gcall_{call_idx}", fc["name"], fc.get("args") or {})
                            call_idx += 1
            yield usage
            yield Done("tool_calls" if saw_calls else "stop")
        except requests.RequestException as e:
            yield ErrorEvent(f"Stream error: {e.__class__.__name__}")
```

"""Ollama native /api/chat client (NDJSON streaming)."""
from __future__ import annotations

import json
from typing import Iterator

import requests

from .base import Done, ErrorEvent, Event, Message, TextDelta, ToolCall, Usage


def messages_to_ollama(messages: list[Message]) -> list[dict]:
    out: list[dict] = []
    for m in messages:
        if m.role == "tool":
            out.append({"role": "tool", "content": str(m.content)})
        elif m.role == "assistant" and m.tool_calls:
            msg: dict = {"role": "assistant", "content": str(m.content or ""),
                         "tool_calls": [{"function": {"name": tc["name"], "arguments": tc["args"]}}
                                        for tc in m.tool_calls]}
            out.append(msg)
        else:
            msg = {"role": m.role, "content": m.content}
            if m.images:
                import base64
                msg["images"] = [base64.b64encode(png).decode() for png in m.images]
            out.append(msg)
    return out


class OllamaProvider:
    def __init__(self, name: str = "ollama", base_url: str = "http://localhost:11434",
                 model: str = "", timeout: int = 120):
        self.name = name
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.timeout = timeout

    def available(self) -> bool:
        try:
            return requests.get(f"{self.base_url}/api/version", timeout=2).ok
        except Exception:
            return False

    def list_models(self) -> list[str]:
        try:
            r = requests.get(f"{self.base_url}/api/tags", timeout=self.timeout)
            r.raise_for_status()
            return sorted(m["name"] for m in r.json().get("models", []))
        except Exception:
            return []

    def stream(self, messages: list[Message], tools: list[dict] | None = None) -> Iterator[Event]:
        payload: dict = {"model": self.model, "stream": True,
                         "messages": messages_to_ollama(messages)}
        if tools:
            payload["tools"] = tools
        try:
            resp = requests.post(f"{self.base_url}/api/chat", json=payload,
                                 timeout=self.timeout, stream=True)
        except requests.RequestException as e:
            yield ErrorEvent(f"Connection error ({self.base_url}): {e.__class__.__name__}")
            return
        if resp.status_code != 200:
            yield ErrorEvent(f"HTTP {resp.status_code}: {resp.text[:300]}")
            return
        saw_calls = False
        call_n = 0
        try:
            for line in resp.iter_lines(decode_unicode=True):
                if not line:
                    continue
                try:
                    chunk = json.loads(line)
                except json.JSONDecodeError:
                    continue
                msg = chunk.get("message") or {}
                if msg.get("content"):
                    yield TextDelta(msg["content"])
                for tc in msg.get("tool_calls") or []:
                    fn = tc.get("function") or {}
                    saw_calls = True
                    yield ToolCall(f"ocall_{call_n}", fn.get("name", ""), fn.get("arguments") or {})
                    call_n += 1
                if chunk.get("done"):
                    yield Usage(chunk.get("prompt_eval_count", 0), chunk.get("eval_count", 0))
                    break
            yield Done("tool_calls" if saw_calls else "stop")
        except requests.RequestException as e:
            yield ErrorEvent(f"Stream error: {e.__class__.__name__}")
```

Ollama returns no tool-call ids, so `stream()` synthesizes deterministic `ocall_N` ids (unique within one stream).

from __future__ import annotations

import configparser
import os

DEFAULT_DIR = os.path.expanduser("~/.config/linchat")
DEFAULT_PATH = os.path.join(DEFAULT_DIR, "config.ini")

ENV_PROVIDERS = [
    # (env var, name, type, base_url)
    ("OPENAI_API_KEY", "openai", "openai", "https://api.openai.com/v1"),
    ("ANTHROPIC_API_KEY", "anthropic", "anthropic", ""),
    ("GEMINI_API_KEY", "gemini", "gemini", ""),
    ("GOOGLE_API_KEY", "gemini", "gemini", ""),

    ("XAI_API_KEY", "xai", "openai", "https://api.x.ai/v1"),
    ("DEEPSEEK_API_KEY", "deepseek", "openai", "https://api.deepseek.com/v1"),
    ("MISTRAL_API_KEY", "mistral", "openai", "https://api.mistral.ai/v1"),
]

    "General": {"default_provider": "", "max_tool_hops": "8", "request_timeout": "60"},
    "Tools": {"shell": "ask", "launch": "ask", "screenshot_cmd": "auto"},
    "Agents": {"disabled": ""},
}


class Config:
    def __init__(self, path: str, data: configparser.ConfigParser):
        self.path = path
        self.data = data

    def providers(self) -> dict[str, dict]:
        out = {}
        for section in self.data.sections():
            if section.startswith("provider:"):
                name = section.split(":", 1)[1]
                out[name] = dict(self.data[section])
        return out

    def env_discovered(self) -> list[dict]:
        env = self._env
        found = []
        for var, name, ptype, base_url in ENV_PROVIDERS:
            if env.get(var) and not any(p == name for p in self.providers()):
                found.append({"name": name, "type": ptype, "base_url": base_url,
                              "api_key": env[var], "model": ""})
        ollama_host = env.get("OLLAMA_HOST")
        if ollama_host and "ollama" not in self.providers():
            found.append({"name": "ollama", "type": "ollama",
                          "base_url": ollama_host, "api_key": "", "model": ""})
        return found

    def get(self, key: str, default: str = "") -> str:
        return self.data.get("General", key, fallback=default)

    def tools_policy(self, tool: str) -> str:
        val = self.data.get("Tools", tool, fallback="ask")
        return val if val in ("ask", "always", "never") else "ask"

    def agent_overrides(self) -> dict:
        section = {k: v for k, v in self.data["Agents"].items()} if self.data.has_section("Agents") else {}
        out = {"disabled": [s.strip() for s in section.pop("disabled", "").split(",") if s.strip()]}
        out.update(section)
        return out

    def set_default_provider(self, name: str) -> None:
        self.data["General"]["default_provider"] = name
        self.save()

    def save(self) -> None:
        os.makedirs(os.path.dirname(self.path), exist_ok=True)
        with open(self.path, "w") as f:
            self.data.write(f)

    _env: dict = {}


def _migrate_v1(data: configparser.ConfigParser, path: str) -> None:
    if not data.has_section("API"):
        return
    backup = path + ".v1.bak"
    if os.path.exists(path) and not os.path.exists(backup):
        with open(path) as src, open(backup, "w") as dst:
            dst.write(src.read())
    api = data["API"]
    base_url = api.get("endpoint", "https://api.openai.com/v1/chat/completions").strip()
    if base_url.endswith("/chat/completions"):
        base_url = base_url[: -len("/chat/completions")]
    data["provider:openai"] = {
        "type": "openai",
        "api_key": api.get("api_key", "").strip(),
        "base_url": base_url,
        "model": api.get("model", "").strip(),
    }
    data.remove_section("API")


def load_config(path: str | None = None, env: dict | None = None) -> Config:
    path = path or DEFAULT_PATH
    data = configparser.ConfigParser()
    for section, values in DEFAULTS.items():
        data[section] = dict(values)
    if os.path.exists(path):
        data.read(path)
        _migrate_v1(data, path)
    cfg = Config(path, data)
    cfg._env = env if env is not None else dict(os.environ)
    if data.get("General", "default_provider", fallback="") == "":
        pass  # registry picks first discovered; written lazily on user choice
    return cfg
```

Careful: `load_config` writes nothing to disk on first run (unlike v1) except migration backup. If the file doesn't exist, defaults live in memory; the file materializes on first `save()`. The migration backup must happen BEFORE `data.read`? No — read first (so backup copies the original v1 bytes), which the order above achieves.

```python
"""CLI agents (opencode, claude-code, codex, gemini, pi, crush) as providers.

Headless invocations verified on 2026-10-19:
  opencode run <msg> | claude -p <msg> --output-format stream-json |
  codex exec <prompt> | gemini -p <prompt> | crush run <prompt...>
`pi` detection uses a --help marker because /usr/bin/pi may be GNU pi
(the constant calculator).
"""
from __future__ import annotations

import dataclasses
import json
import shlex
import subprocess
from typing import Iterator

from .base import Done, ErrorEvent, Event, Message, TextDelta


@dataclasses.dataclass
class AgentDef:
    name: str
    binary: str
    args: list[str]
    parser: str = "plain"       # plain | claude-json
    help_marker: str = ""       # required substring of `<binary> --help`


AGENTS: list[AgentDef] = [
    AgentDef("opencode", "opencode", ["run", "{prompt}"]),
    AgentDef("claude-code", "claude", ["-p", "{prompt}", "--output-format", "stream-json"],
             parser="claude-json"),
    AgentDef("codex", "codex", ["exec", "{prompt}"]),
    AgentDef("gemini", "gemini", ["-p", "{prompt}"]),
    AgentDef("pi", "pi", ["{prompt}"], help_marker="agent"),
    AgentDef("crush", "crush", ["run", "{prompt}"]),
]


def _run_help(binary: str) -> str:
    try:
        return subprocess.run([binary, "--help"], capture_output=True, text=True,
                              timeout=15).stdout
    except Exception:
        return ""


def detect_agents(overrides: dict | None = None, which=None) -> list[AgentDef]:
    import shutil
    which = which or shutil.which
    ov = overrides or {}
    disabled = ov.get("disabled", [])
    found = []
    for d in AGENTS:
        if d.name in disabled:
            continue
        binary = ov.get(f"{d.name}_bin", d.binary)
        path = which(binary)
        if not path:
            continue
        if d.help_marker and d.help_marker not in _run_help(path):
            continue
        args = shlex.split(ov[f"{d.name}_args"]) if f"{d.name}_args" in ov else d.args
        found.append(dataclasses.replace(d, binary=path, args=args))
    return found


class AgentProvider:
    def __init__(self, defn: AgentDef, timeout: int = 600):
        self.name = defn.name
        self.defn = defn
        self.timeout = timeout

    def list_models(self) -> list[str]:
        return []

    def stream(self, messages: list[Message], tools: list[dict] | None = None) -> Iterator[Event]:
        prompt = "\n".join(str(m.content) for m in messages if m.role == "user")
        cmd = [self.defn.binary] + [a.replace("{prompt}", prompt) for a in self.defn.args]
        try:
            proc = subprocess.Popen(cmd, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                                    text=True, start_new_session=True)
        except OSError as e:
            yield ErrorEvent(f"Failed to start {self.defn.name}: {e}")
            return
        try:
            for line in proc.stdout:
                if self.defn.parser == "claude-json":
                    yield from self._claude_line(line)
                elif line.strip():
                    yield TextDelta(line)
            proc.wait(timeout=self.timeout)
        except subprocess.TimeoutExpired:
            proc.kill()
            yield ErrorEvent(f"{self.defn.name}: timed out after {self.timeout}s")
            return
        stderr = proc.stderr.read() if proc.stderr else ""
        if proc.returncode != 0:
            yield ErrorEvent(f"{self.defn.name} (exit {proc.returncode}): "
                             f"{shlex.join(cmd)}\n{stderr.strip()[-500:]}")
            return
        yield Done("stop")

    def _claude_line(self, line: str) -> Iterator[Event]:
        try:
            ev = json.loads(line)
        except json.JSONDecodeError:
            return
        if ev.get("type") == "assistant":
            for block in ev.get("message", {}).get("content", []):
                if block.get("type") == "text" and block.get("text"):
                    yield TextDelta(block["text"])
```

```python
"""Assemble the provider list: config sections, env discovery, agents."""
from __future__ import annotations

from typing import Iterator

from ..config import Config
from .agents import AgentDef, AgentProvider, detect_agents
from .anthropic import AnthropicProvider
from .base import ErrorEvent, Event, Message
from .gemini import GeminiProvider
from .ollama import OllamaProvider
from .openai_compat import OpenAICompatProvider


def _keyring_get(name: str) -> str | None:
    try:
        import keyring
        return keyring.get_password("linchat", name)
    except Exception:
        return None


class NullProvider:
    name = "none"

    def list_models(self) -> list[str]:
        return []

    def stream(self, messages, tools=None) -> Iterator[Event]:
        yield ErrorEvent("No providers configured — set an API key in "
                         "~/.config/linchat/config.ini or export e.g. OPENAI_API_KEY")


def _build_hosted(spec: dict, env_keys: dict[str, str], timeout: int) -> object | None:
    ptype = spec.get("type", "openai")
    key = spec.get("api_key", "") or env_keys.get(spec["name"], "") \
        or _keyring_get(spec["name"]) or ""
    base = spec.get("base_url", "")
    model = spec.get("model", "")
    if ptype == "anthropic":
        return AnthropicProvider(spec["name"], key, model, timeout) if key else None
    if ptype == "gemini":
        return GeminiProvider(spec["name"], key, model, timeout) if key else None
    if ptype == "ollama":
        return OllamaProvider(spec["name"], base or "http://localhost:11434", model, timeout)
    # openai / openai-compat
    if not key and "localhost" not in base and "127.0.0.1" not in base:
        return None  # remote endpoints need a key
    return OpenAICompatProvider(spec["name"], key, base or "https://api.openai.com/v1",
                                model, timeout)


class Registry:
    def __init__(self, config: Config):
        self.config = config
        self._providers: list | None = None

    def refresh(self) -> None:
        self._providers = None

    def providers(self) -> list:
        if self._providers is not None:
            return self._providers
        timeout = int(self.config.get("request_timeout", "60"))
        env_specs = self.config.env_discovered()
        env_keys = {s["name"]: s["api_key"] for s in env_specs}
        provs: list = []
        specs: list[dict] = []
        for name, section in self.config.providers().items():
            specs.append({"name": name, **section})
        have = {s["name"] for s in specs}
        for spec in env_specs:
            if spec["name"] not in have:
                specs.append(spec)
                have.add(spec["name"])
        for spec in specs:
            p = _build_hosted(spec, env_keys, timeout)
            if p is not None:
                provs.append(p)
        if "ollama" not in have:
            candidate = OllamaProvider("ollama", timeout=timeout)
            if candidate.available():
                provs.append(candidate)
        for d in detect_agents(self.config.agent_overrides()):
            provs.append(AgentProvider(d, timeout=600))
        self._providers = provs
        return provs

    def get(self, name: str):
        for p in self.providers():
            if p.name == name:
                return p
        return None

    def default(self):
        provs = self.providers()
        wanted = self.config.get("default_provider", "")
        if wanted:
            p = self.get(wanted)
            if p is not None:
                return p
        hosted = [p for p in provs if not isinstance(p, AgentProvider)]
        if hosted:
            return hosted[0]
        agents = [p for p in provs if isinstance(p, AgentProvider)]
        if agents:
            return agents[0]
        return NullProvider()
```

```python
"""Slash-command parsing and autocomplete (execution lives in the tools layer)."""
from __future__ import annotations

import dataclasses


@dataclasses.dataclass
class Command:
    name: str
    args: str    # usage hint shown in autocomplete
    description: str


COMMANDS: list[Command] = [
    Command("run", "cmd", "execute a shell command"),
    Command("open", "name", "launch an app"),
    Command("steam", "query", "search Steam library and store"),
    Command("games", "query", "search all game libraries"),
    Command("find", "pattern", "search files"),
    Command("read", "path", "read a file"),
    Command("clip", "", "read clipboard into context"),
    Command("snap", "", "capture a screen region for the next prompt"),
]

_NAMES = {c.name for c in COMMANDS}


def parse(text: str) -> tuple[str, str] | None:
    if not text.startswith("/"):
        return None
    body = text[1:]
    name, sep, argstr = body.partition(" ")
    if name not in _NAMES:
        return None
    if sep and not argstr.strip() and name != "clip":
        return (name, "")
    return (name, argstr)


def complete(prefix: str) -> list[Command]:
    p = prefix.lstrip("/")
    return [c for c in COMMANDS if c.name.startswith(p)]
```
